2. 图片数据库模型
image/models.py
from sqlalchemy import (
    Column,
    String,
    Text,
    DateTime,
    ForeignKey
)

from datetime import datetime
import uuid

from database.base import Base


class Image(Base):

    __tablename__ = "images"


    id = Column(
        String,
        primary_key=True,
        default=lambda:str(uuid.uuid4())
    )


    customer_id = Column(
        String,
        ForeignKey(
            "customers.id"
        )
    )


    filename = Column(
        String(200)
    )


    file_path = Column(
        Text
    )


    image_type = Column(
        String(50)
    )


    description = Column(
        Text
    )


    uploaded_at = Column(
        DateTime,
        default=datetime.utcnow
    )
3. 图片存储
image/storage.py

负责保存图片文件。

import os
import uuid


UPLOAD_FOLDER = "storage/images"



def save_image(file):


    os.makedirs(

        UPLOAD_FOLDER,

        exist_ok=True

    )


    extension = (

        file.filename

        .split(".")

        [-1]

    )


    filename = (

        str(uuid.uuid4())

        +

        "."

        +

        extension

    )


    path = os.path.join(

        UPLOAD_FOLDER,

        filename

    )


    with open(

        path,

        "wb"

    ) as buffer:


        buffer.write(

            file.file.read()

        )


    return path

5. FastAPI 上传接口
image/api.py
from fastapi import (
    APIRouter,
    UploadFile,
    File,
    Form,
    Depends
)

from database.connection import SessionLocal

from image.storage import save_image

from image.service import create_image


router = APIRouter(
    prefix="/images"
)



def get_db():

    db = SessionLocal()

    try:

        yield db

    finally:

        db.close()



@router.post("/upload")
def upload_image(

    customer_id:str=Form(...),

    image_type:str=Form(None),

    description:str=Form(None),

    file:UploadFile=File(...),

    db=Depends(get_db)

):


    path = save_image(file)


    image = create_image(

        db,

        customer_id,

        file.filename,

        path,

        image_type,

        description

    )


    return {

        "message":

        "upload success",

        "image_id":

        image.id

    }
6. 数据库保存服务
image/service.py
from image.models import Image



def create_image(

    db,

    customer_id,

    filename,

    path,

    image_type,

    description

):


    image = Image(

        customer_id=customer_id,

        filename=filename,

        file_path=path,

        image_type=image_type,

        description=description

    )


    db.add(image)

    db.commit()

    db.refresh(image)


    return image
