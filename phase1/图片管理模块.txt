2. 图片存储模块
image/storage.py

负责保存文件。

import os
import uuid


IMAGE_FOLDER = "storage/images"



def save_image(file):


    os.makedirs(
        IMAGE_FOLDER,
        exist_ok=True
    )


    ext = file.filename.split(".")[-1]


    filename = (
        str(uuid.uuid4())
        +
        "."
        +
        ext
    )


    path=os.path.join(
        IMAGE_FOLDER,
        filename
    )


    with open(
        path,
        "wb"
    ) as buffer:

        buffer.write(
            file.file.read()
        )


    return path

4. 图片业务逻辑
image/service.py
from database.models import Image



def create_image_record(
        db,
        data,
        path,
        filename
):


    image=Image(

        customer_id=data.customer_id,

        visit_id=data.visit_id,

        filename=filename,

        file_path=path,

        image_type=data.image_type,

        description=data.description

    )


    db.add(image)

    db.commit()

    db.refresh(image)


    return image



def get_customer_images(
        db,
        customer_id
):


    return db.query(
        Image
    ).filter(

        Image.customer_id
        ==
        customer_id

    ).all()



def delete_image(
        db,
        image
):

    db.delete(image)

    db.commit()
5. 上传 API
image/api.py
from fastapi import (
    APIRouter,
    UploadFile,
    File,
    Form,
    Depends
)


from database.connection import SessionLocal

from image.storage import save_image

from image.service import create_image_record


router=APIRouter(
    prefix="/images"
)



def get_db():

    db=SessionLocal()

    try:

        yield db

    finally:

        db.close()



@router.post("/upload")
def upload_image(

    customer_id:str=Form(...),

    visit_id:str=None,

    image_type:str=Form(None),

    description:str=Form(None),

    file:UploadFile=File(...),

    db=Depends(get_db)

):


    path=save_image(file)


    data={

        "customer_id":customer_id,

        "visit_id":visit_id,

        "image_type":image_type,

        "description":description

    }


    return create_image_record(

        db,

        data,

        path,

        file.filename

    )
