1. Customer 数据表

database/models.py

from sqlalchemy import (
    Column,
    String,
    Boolean,
    DateTime
)

from datetime import datetime
import uuid

from database.base import Base


class Customer(Base):

    __tablename__ = "customers"


    id = Column(
        String,
        primary_key=True,
        default=lambda:str(uuid.uuid4())
    )


    name = Column(
        String(100),
        nullable=False
    )


    gender = Column(
        String(20)
    )


    birthday = Column(
        String(20)
    )


    phone = Column(
        String(50)
    )


    email = Column(
        String(200),
        unique=True
    )


    address = Column(
        String(300)
    )


    telegram_id = Column(
        String(100)
    )


    whatsapp = Column(
        String(50)
    )


    notification_enabled = Column(
        Boolean,
        default=True
    )


    created_at = Column(
        DateTime,
        default=datetime.utcnow
    )

3. 客户服务层
customer/service.py

这里负责业务逻辑。

from database.models import Customer



def create_customer(
        db,
        data
):


    customer = Customer(

        name=data.name,

        gender=data.gender,

        birthday=data.birthday,

        phone=data.phone,

        email=data.email,

        address=data.address,

        telegram_id=data.telegram_id,

        whatsapp=data.whatsapp

    )


    db.add(customer)

    db.commit()

    db.refresh(customer)


    return customer



def get_customers(db):


    return db.query(
        Customer
    ).all()



def get_customer(
        db,
        customer_id
):


    return db.query(
        Customer
    ).filter(

        Customer.id == customer_id

    ).first()



def update_customer(
        db,
        customer,
        data
):


    if data.phone:

        customer.phone=data.phone


    if data.email:

        customer.email=data.email


    if data.address:

        customer.address=data.address



    db.commit()

    db.refresh(customer)


    return customer



def delete_customer(
        db,
        customer
):


    db.delete(customer)

    db.commit()
4. API接口
customer/api.py
from fastapi import APIRouter, Depends

from database.connection import SessionLocal

from customer.schema import *

from customer.service import *


router = APIRouter(
    prefix="/customers"
)



def get_db():

    db=SessionLocal()

    try:

        yield db

    finally:

        db.close()



# 新增客户

@router.post("/")
def add_customer(
    data:CustomerCreate,
    db=Depends(get_db)
):

    return create_customer(
        db,
        data
    )



# 查询全部客户

@router.get("/")
def list_customer(
    db=Depends(get_db)
):

    return get_customers(db)



# 查询单个客户

@router.get("/{id}")
def detail_customer(
    id:str,
    db=Depends(get_db)
):

    return get_customer(
        db,
        id
    )



# 修改

@router.put("/{id}")
def edit_customer(
    id:str,
    data:CustomerUpdate,
    db=Depends(get_db)
):

    customer=get_customer(
        db,
        id
    )

    return update_customer(
        db,
        customer,
        data
    )



# 删除

@router.delete("/{id}")
def remove_customer(
    id:str,
    db=Depends(get_db)
):

    customer=get_customer(
        db,
        id
    )

    delete_customer(
        db,
        customer
    )

    return {
        "message":"deleted"
    }
